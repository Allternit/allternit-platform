cmake_minimum_required(VERSION 3.22)
project(allternit_zoom_sidecar VERSION 0.1.0 LANGUAGES CXX)

# Zoom Meeting SDK for Linux, unpacked (h/, libmeetingsdk.so, qt_libs/). Not in git:
# it is downloaded from our Marketplace app's Embed page (see README.md "Getting the SDK").
set(ZOOM_SDK_DIR "${CMAKE_CURRENT_SOURCE_DIR}/vendor/zoom-sdk" CACHE PATH "Unpacked Zoom Meeting SDK for Linux")

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE RelWithDebInfo)
endif()

if(NOT EXISTS "${ZOOM_SDK_DIR}/h/zoom_sdk.h")
  message(FATAL_ERROR "Zoom SDK headers not found at ${ZOOM_SDK_DIR}/h. Unpack the SDK there or pass -DZOOM_SDK_DIR=... (README.md).")
endif()

find_package(PkgConfig REQUIRED)
pkg_check_modules(GLIB REQUIRED IMPORTED_TARGET glib-2.0)
find_package(nlohmann_json 3.10 REQUIRED)
find_package(Threads REQUIRED)

# The SDK library's SONAME is libmeetingsdk.so.1 but the tarball ships libmeetingsdk.so;
# Zoom's sample copies it. Do the same in the build tree.
if(NOT EXISTS "${ZOOM_SDK_DIR}/libmeetingsdk.so.1" AND EXISTS "${ZOOM_SDK_DIR}/libmeetingsdk.so")
  file(COPY_FILE "${ZOOM_SDK_DIR}/libmeetingsdk.so" "${ZOOM_SDK_DIR}/libmeetingsdk.so.1")
endif()

# Enum names move between SDK versions. Probe each one against the real headers and
# generate HAVE_<NAME> macros, so bot.cpp compiles on any 6.x/7.x SDK and names the
# codes that exist (the adapter maps them by name, index.ts mapJoinFailure).
set(ZOOM_ENUM_NAMES
  MEETING_FAIL_NETWORK_ERR MEETING_FAIL_RECONNECT_ERR MEETING_FAIL_MMR_ERR MEETING_FAIL_PASSWORD_ERR
  MEETING_FAIL_SESSION_ERR MEETING_FAIL_MEETING_OVER MEETING_FAIL_MEETING_NOT_START MEETING_FAIL_MEETING_NOT_EXIST
  MEETING_FAIL_MEETING_USER_FULL MEETING_FAIL_CLIENT_INCOMPATIBLE MEETING_FAIL_CONFLOCKED
  MEETING_FAIL_MEETING_RESTRICTED MEETING_FAIL_MEETING_RESTRICTED_JBH MEETING_FAIL_CANNOT_START_TOKENEXPIRE
  MEETING_FAIL_BLOCKED_BY_ACCOUNT_ADMIN MEETING_FAIL_NEED_SIGN_IN_FOR_PRIVATE_MEETING
  MEETING_FAIL_APP_PRIVILEGE_TOKEN_ERROR MEETING_FAIL_AUTHORIZED_USER_NOT_INMEETING
  MEETING_FAIL_ON_BEHALF_TOKEN_CONFLICT_LOGIN_ERROR MEETING_FAIL_JMAK_USER_EMAIL_NOT_MATCH
  EndMeetingReason_None EndMeetingReason_KickByHost EndMeetingReason_EndByHost EndMeetingReason_JBHTimeOut
  EndMeetingReason_NoAttendee EndMeetingReason_HostStartAnotherMeeting EndMeetingReason_FreeMeetingTimeOut
  EndMeetingReason_NetworkBroken
  RequestLocalRecording_Granted RequestLocalRecording_Denied RequestLocalRecording_Timeout
)
include(CheckCXXSourceCompiles)
set(CMAKE_REQUIRED_INCLUDES "${ZOOM_SDK_DIR}/h")
set(CMAKE_REQUIRED_QUIET ON)
set(_gen "// Generated by CMakeLists.txt from the SDK headers. Do not edit.\n#pragma once\n")
foreach(_name IN LISTS ZOOM_ENUM_NAMES)
  check_cxx_source_compiles("
#include \"zoom_sdk.h\"
#include \"meeting_service_interface.h\"
#include \"meeting_service_components/meeting_recording_interface.h\"
using namespace ZOOMSDK;
int main() { return static_cast<int>(${_name}) == -123456 ? 1 : 0; }
" HAVE_${_name})
  if(HAVE_${_name})
    string(APPEND _gen "#define HAVE_${_name} 1\n")
  else()
    message(STATUS "Zoom SDK lacks ${_name}; that code is reported by number only")
  endif()
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/gen/zoom_names.gen.h" "${_gen}")

add_executable(zoom-sidecar src/main.cpp src/ipc.cpp src/bot.cpp)
target_include_directories(zoom-sidecar PRIVATE "${ZOOM_SDK_DIR}/h" "${CMAKE_CURRENT_BINARY_DIR}/gen")
target_link_directories(zoom-sidecar PRIVATE "${ZOOM_SDK_DIR}")
target_link_libraries(zoom-sidecar PRIVATE meetingsdk PkgConfig::GLIB nlohmann_json::nlohmann_json Threads::Threads)
target_compile_options(zoom-sidecar PRIVATE -Wall -Wextra -Wno-unused-parameter)
# Installed layout: <prefix>/zoom-sidecar next to <prefix>/lib/{libmeetingsdk.so*, qt_libs/}.
set_target_properties(zoom-sidecar PROPERTIES
  BUILD_RPATH "${ZOOM_SDK_DIR};${ZOOM_SDK_DIR}/qt_libs;${ZOOM_SDK_DIR}/qt_libs/Qt/lib"
  INSTALL_RPATH "\$ORIGIN/lib;\$ORIGIN/lib/qt_libs;\$ORIGIN/lib/qt_libs/Qt/lib")

install(TARGETS zoom-sidecar RUNTIME DESTINATION .)
install(DIRECTORY "${ZOOM_SDK_DIR}/" DESTINATION lib PATTERN "h" EXCLUDE)
